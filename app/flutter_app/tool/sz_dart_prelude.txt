// JS typed-array stores coerce numbers; Float64Array also accepts integer values.
class _NumericArray extends ListBase<dynamic> {
  _NumericArray(this.kind, dynamic lengthOrData)
    : values = lengthOrData is num ? List<num>.filled(lengthOrData.toInt(), 0)
      : List<num>.from(lengthOrData as Iterable);
  final String kind;
  final List<num> values;
  @override int get length => values.length;
  @override set length(int n) => values.length=n;
  @override dynamic operator [](int i) => values[i];
  @override void operator []=(int i, dynamic v) {
    final n=v as num;
    if(kind=='Uint8ClampedArray') {
      final f=n.toDouble().clamp(0,255), low=f.floor(), part=f-low;
      values[i]=part>.5 || (part==.5 && low.isOdd) ? low+1 : low;
    } else if(kind=='Uint8Array') { values[i]=n.toInt() & 255; }
    else { values[i]=n.toDouble(); }
  }
}

/// Frozen YRD0370 V7.5 seed, structure refinement and sparkle/chroma cleanup.
Future<Uint8List> quantizeSz(Uint8List rgba,int width,int height,{
  void Function()? checkCancelled,void Function(double)? onProgress,
}) async {
  checkCancelled?.call();
  final engine=_SzEngine(checkCancelled,onProgress);
  final input=<String,dynamic>{'width':width,'height':height,'data':rgba};
  final output=await engine.szEnhancedDither(input);
  final data=output['data'];
  const display=[[0,0,0],[255,255,255],[255,255,0],[255,0,0],[0,255,0],[0,0,255]];
  const toFilm=[0,1,2,3,5,4];
  final indices=Uint8List(width*height);
  for(var i=0;i<indices.length;i++) {
    final p=i*4;
    final index=display.indexWhere((c)=>c[0]==data[p] && c[1]==data[p+1] && c[2]==data[p+2]);
    indices[i]=toFilm[index];
    rgba[p]=data[p];rgba[p+1]=data[p+1];rgba[p+2]=index==4?20:data[p+2];
    if(index==4){rgba[p]=41;rgba[p+1]=204;}
    rgba[p+3]=255;
  }
  checkCancelled?.call();onProgress?.call(.9);
  return indices;
}
